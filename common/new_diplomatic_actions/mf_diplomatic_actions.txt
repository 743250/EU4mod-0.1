	category = shogun
	alert_index = 17
	alert_tooltip = xuanxiafengxing_tooltip
	require_acceptance = no

	is_visible = {
		ROOT = {
			tag = event_target:zhengyidajiangjun
			has_global_flag = shogun_weixin.d_global_flag
		}
		FROM = {
			is_subject_of = ROOT
			NOT = { has_country_flag = mf_xiayi_target }
			NOT = { has_country_flag = mf_tianhuang_target }
		}
	}

	is_allowed = {
		custom_trigger_tooltip = {
			tooltip = mf_xuanxiafengxing_authority_tt
			1020 = {
				check_variable = {
					which = shogun_value
					value = 10
				}
			}
		}
		custom_trigger_tooltip = {
			tooltip = mf_xuanxiafengxing_cooldown_tt
			OR = {
				had_country_flag = {
					flag = xuanxiafengxing_flag
					days = 3650
				}
				NOT = { has_country_flag = xuanxiafengxing_flag }
			}
		}
	}

	on_accept = {
		if = {
			limit = { has_saved_global_event_target = tianhuang }
			event_target:tianhuang = {
				clr_country_flag = mf_tianhuang_target
				clear_global_event_target = tianhuang
			}
		}
		FROM = {
			save_global_event_target_as = tianhuang
			set_country_flag = mf_tianhuang_target
		}
		clr_country_flag = xuanxiafengxing_flag
		set_country_flag = xuanxiafengxing_flag
		mf_change_shogun_value = {
			value = -10
		}
	}

	ai_will_do = {
		always = no
	}
}

mf_return_shogunate_province = {
	category = shogun
	alert_index = 42
	alert_tooltip = mf_return_shogunate_province_tooltip
	require_acceptance = yes

	is_visible = {
		tag = event_target:zhengyidajiangjun
		has_global_flag = shogun_weixin.b_global_flag
	}

	is_allowed = {
		is_at_war = no
		FROM = {
			any_owned_province = {
				has_province_flag = mf_shogunate_province
				NOT = { is_core = FROM }
			}
			is_at_war = no
		}
	}

	on_accept = {
		hidden_effect = {
			FROM = {
				save_event_target_as = mf_illegal_land_holder
				random_owned_province = {
					limit = {
						has_province_flag = mf_shogunate_province
						NOT = { is_core = event_target:mf_illegal_land_holder }
					}
					save_event_target_as = mf_return_shogunate_province
				}
			}
			event_target:mf_return_shogunate_province = {
				remove_province_modifier = mf_unlawful_daimyo_land
				if = {
					limit = {
						any_core_country = { sort_of_mf = yes }
					}
					random_core_country = {
						limit = { sort_of_mf = yes }
						save_event_target_as = mf_return_shogunate_province_owner
					}
					cede_province = event_target:mf_return_shogunate_province_owner
					province_event = { id = mf_daimyo.12 }
				}
				else = {
					cede_province = event_target:zhengyidajiangjun
				}
			}
		}
		reverse_add_opinion = {
			who = FROM
			modifier = opinion_mf_demanded_unlawful_land
		}
		mf_change_shogun_value = { value = 2 }
	}
	on_decline = {
		FROM = {
			save_event_target_as = mf_illegal_land_holder
			set_country_flag = mf_refused_return_shogunate_province
			# 拒绝归还时，把目标持有的全部无本国核心登记领地标记为非法领土。

				limit = {
					has_province_flag = mf_shogunate_province
					NOT = { is_core = event_target:mf_illegal_land_holder }
				}
				remove_province_modifier = mf_unlawful_daimyo_land
				add_province_modifier = {
					name = mf_unlawful_daimyo_land
					duration = -1
				}
			}
		}
		add_casus_belli = {
			target = FROM
			type = cb_mf_return_daimyo_land
			months = 60
		}
		reverse_add_opinion = {
			who = FROM
			modifier = opinion_mf_demanded_unlawful_land
		}
		mf_change_shogun_value = { value = -2 }
	}

	ai_acceptance = {
		add_entry = {
			name = mf_shogunate_order_base_reluctance
			export_to_variable = { variable_name = ai_value value = -75 }
		}
		add_entry = {
			name = mf_shogunate_order_negative_prestige
			limit = { NOT = { prestige = 0 } }
			export_to_variable = {
				variable_name = ai_value
				value = prestige
				who = ROOT
			}
			multiply_variable = { which = ai_value value = 0.2 }
			round_variable = { which = ai_value value = 0 }
		}
		add_entry = {
			name = mf_shogunate_order_previously_demanded
			limit = {
				reverse_has_opinion_modifier = {
					who = FROM
					modifier = opinion_mf_demanded_unlawful_land
				}
			}
			export_to_variable = { variable_name = ai_value value = -100 }
		}
		add_entry = {
			name = mf_shogunate_diplomatic_acceptance
			export_to_variable = {
				variable_name = ai_value
				value = modifier:mf_shogunate_diplomatic_acceptance
				who = ROOT
			}
			round_variable = { which = ai_value value = 0 }
		}
		add_entry = {
			name = AI_DIPLOMATIC_REPUTATION
			export_to_variable = {
				variable_name = ai_value
				value = trigger_value:diplomatic_reputation
			}
			multiply_variable = { which = ai_value value = 2 }
			round_variable = { which = ai_value value = 0 }
		}
		add_entry = {
			name = mf_return_land_outside_shogunate
			export_to_variable = { variable_name = ai_value value = -50 }
		}
		add_entry = {
			name = mf_shogunate_order_fear

				variable_name = ai_value
				value = manpower
				who = ROOT
				with = FROM
			}
			export_to_variable = {
				variable_name = mf_target_manpower
				value = manpower
				who = FROM
			}
			subtract_variable = { which = ai_value which = mf_target_manpower }
			export_to_variable = {
				variable_name = mf_shogun_discipline

			}

			change_variable = { which = ai_value which = mf_shogun_discipline }
			multiply_variable = { which = ai_value value = 0.4 }
			round_variable = { which = ai_value value = 0 }

			name = DIPLO_OPPINION

				variable_name = ai_value

				who = FROM
				with = ROOT
			}

			round_variable = { which = ai_value value = 0 }
		}
		add_entry = {
			name = mf_return_land_unfriendly_attitude
			limit = {
				FROM = {
					OR = {
						ai_attitude = { who = ROOT attitude = attitude_rivalry }

	alert_index = 23
	alert_tooltip = mf_lecture_daimyo_tooltip
	require_acceptance = no

	is_visible = {
		has_reform = shogunate
		has_country_flag = mf_shogun_unlock_lecture_grace
		FROM = {
			is_subject_of = ROOT
			OR = {
				is_subject_of_type = daimyo_vassal
				is_subject_of_type = mf_daimyo_vassal
			}
		}
	}

	is_allowed = {
		NOT = { war_with = FROM }
		custom_trigger_tooltip = {
			tooltip = mf_shogunate_order_authority_lecture_tt
			OR = {
				AND = {
					has_country_flag = mf_shogun_lecture_grace_upgraded
					1020 = {
						check_variable = {
							which = shogun_value
							value = 3
						}
					}
				}
				AND = {
					NOT = { has_country_flag = mf_shogun_lecture_grace_upgraded }
					1020 = {
						check_variable = {
							which = shogun_value
							value = 5
						}
					}
				}
			}
		}
		FROM = {
			NOT = { has_country_modifier = mf_shogunate_lecture }
			NOT = { has_country_modifier = mf_shogunate_grace }
			custom_trigger_tooltip = {
				tooltip = mf_lecture_daimyo_cooldown_tt
				OR = {
					NOT = { has_country_flag = mf_recently_lectured_by_shogun }
					had_country_flag = {
						flag = mf_recently_lectured_by_shogun
						days = 1825
					}
				}
			}
		}
	}

	on_accept = {
		FROM = {
			clr_country_flag = mf_recently_lectured_by_shogun
			set_country_flag = mf_recently_lectured_by_shogun
			remove_country_modifier = mf_shogunate_grace
			add_country_modifier = {
				name = mf_shogunate_lecture
				duration = 3650
			}
			add_liberty_desire = -10
			add_prestige = -5
			add_legitimacy = -5
			add_opinion = {
				who = ROOT
				modifier = opinion_mf_lectured_by_shogun
			}
		}
		if = {
			limit = {
				has_country_flag = mf_shogun_lecture_grace_upgraded
			}
			mf_change_shogun_value = {
				value = -3
			}
		}
		else = {
			mf_change_shogun_value = {
				value = -5
			}
		}
	}

	ai_will_do = {
		always = yes
	}
}
