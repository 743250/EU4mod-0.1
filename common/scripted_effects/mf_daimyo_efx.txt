########## 通用费用处理 ##########
treasury_reduce = {
	hidden_effect = {
		while = {
			# 先按整数扣款，再把剩余小数放大到分位处理。
			limit = {
				check_variable = {
					which = $variable$
					value = 1
				}
			}
			add_treasury = -1
			subtract_variable = {
				which = $variable$
				value = 1
			}
		}
		multiply_variable = {
			which = $variable$
			value = 100
		}
		while = {
			limit = {
				check_variable = {
					which = $variable$
					value = 1
				}
			}
			add_treasury = -0.01
			subtract_variable = {
				which = $variable$
				value = 1
			}
		}
	}
}
########## 御统与幕府外交效果 ##########
mf_change_shogun_value = {
	# 御统显示与缩放都读 1020；所有增减统一从这里走，防止越过 0-100。
	hidden_effect = {
		1020 = {
			change_variable = {
				which = shogun_value
				value = $value$
			}
			if = {
				limit = {
					NOT = {
						check_variable = {
							which = shogun_value
							value = 0
						}
					}
				}
				set_variable = {
					which = shogun_value
					value = 0
				}
				set_variable = {
					which = shogun_value_Profix
					value = 0
				}
			}
			if = {
				limit = {
					check_variable = {
						which = shogun_value
						value = 100
					}
				}
				set_variable = {
					which = shogun_value
					value = 100
				}
				set_variable = {
					which = shogun_value_Profix
					value = 0
				}
			}
			set_variable = {
				which = shogun_value_display
				which = shogun_value
			}
			set_variable = {
				which = shogun_value_display_fraction
				which = shogun_value_Profix
			}
			multiply_variable = {
				which = shogun_value_display_fraction
				value = 0.01
			}
			change_variable = {
				which = shogun_value_display
				which = shogun_value_display_fraction
			}
		}
	}
}

mf_force_partition_daimyo_effect = {
	# 强制分割会一次性归还目标持有的已灭亡旧家核心，并由 mf_daimyo.12 重新纳入幕府。
	save_event_target_as = mf_partition_daimyo_target
	every_owned_province = {
		limit = {
			is_capital = no
			any_core_country = {
				NOT = { tag = event_target:mf_partition_daimyo_target }
				NOT = { tag = ROOT }
			}
		}
		random_core_country = {
			limit = {
				NOT = { tag = event_target:mf_partition_daimyo_target }
				NOT = { tag = ROOT }
			}
			PREV = {
				remove_province_modifier = mf_unlawful_daimyo_land
				cede_province = PREV
				province_event = {
					id = mf_daimyo.12
				}
			}
			add_opinion = {
				who = ROOT
				modifier = opinion_mf_partition_restored_land
			}
		}
	}
	add_country_modifier = {
		name = mf_shogunate_partitioned
		duration = 3650
	}
}

mf_appoint_executor_daimyo_effect = {
	create_subject = {
		subject_type = mf_daimyo_vassal
		subject = FROM
	}
	FROM = {
		add_government_reform = mf_daimyo
		clr_country_flag = mf_shogunate_executor
		set_country_flag = mf_shogunate_executor
		add_country_modifier = {
			name = mf_shogunate_executor
			duration = -1
		}
		add_liberty_desire = -25
		add_opinion = {
			who = ROOT
			modifier = opinion_mf_appointed_executor
		}
	}
}
