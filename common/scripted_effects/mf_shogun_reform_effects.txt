########## 幕府大型计划共用效果 ##########

mf_calculate_monthly_reform_authority_effect = {
	# 维新月度统御分项在京都变量中单独计算，避免与普通大名数量混在一起。
	set_variable = {
		which = shogun_monthly_value_9
		value = 0.00
	}
	set_variable = {
		which = shogun_monthly_value_10
		value = 0.00
	}
	if = {
		limit = { has_global_flag = shogun_weixin.e_global_flag }
		event_target:zhengyidajiangjun = {
			every_subject_country = {
				limit = { is_subject_of_type = mf_daimyo_vassal }
				1020 = {
					change_variable = {
						which = shogun_monthly_value_9
						value = 0.02
					}
				}
			}
		}
	}
	if = {
		limit = {
			has_global_flag = shogun_weixin.i_global_flag
			has_saved_global_event_target = xiayi
			event_target:xiayi = {
				exists = yes
				has_country_flag = mf_xiayi_target
				NOT = { sort_of_mf = yes }
				NOT = { has_country_flag = mf_tianhuang_target }
			}
			event_target:zhengyidajiangjun = {
				war_with = event_target:xiayi
			}
		}
		change_variable = {
			which = shogun_monthly_value_10
			value = 0.02
		}
	}
}

mf_add_permanent_claims_on_japan_region_effect = {
	# 给日本地区未核心、未永久宣称省份添加永久宣称。
	japan_region = {
		limit = {
			NOT = { is_core = ROOT }
			NOT = { is_permanent_claim = ROOT }
		}
		add_permanent_claim = ROOT
	}
}

mf_add_permanent_claim_on_ryukyu_effect = {
	# 琉球方向只取冲绳单省，避免误覆盖越南区域。
	if = {
		limit = {
			1015 = {
				NOT = { is_core = ROOT }
				NOT = { is_permanent_claim = ROOT }
			}
		}
		1015 = {
			add_permanent_claim = ROOT
		}
	}
}

mf_add_permanent_claims_on_korea_region_effect = {
	# 给朝鲜地区未核心、未永久宣称省份添加永久宣称。
	korea_region = {
		limit = {
			NOT = { is_core = ROOT }
			NOT = { is_permanent_claim = ROOT }
		}
		add_permanent_claim = ROOT
	}
}

mf_complete_sankin_reform_effect = {
	# 玩家和 AI 共用同一参勤改革效果。
	set_global_flag = shogun_weixin.h_global_flag
	mf_change_shogun_value = { value = -70 }
	remove_country_modifier = shogun_yuling_b
	remove_country_modifier = overlord_sankin_kotai
	add_country_modifier = {
		name = jiujizhengshi_2
		duration = -1
		hidden = yes
	}
	every_subject_country = {
		limit = {
			sort_of_mf = yes
			NOT = { has_reform = shogunate }
		}
		remove_country_modifier = subject_sankin_kotai
		add_country_modifier = {
			name = jiujizhengshi_1
			duration = -1
			hidden = yes
		}
		every_subject_country = {
			limit = {
				sort_of_mf = yes
				NOT = { has_reform = shogunate }
			}
			remove_country_modifier = subject_sankin_kotai
			add_country_modifier = {
				name = jiujizhengshi_1
				duration = -1
				hidden = yes
			}
		}
	}
}
