mf_refresh_tianhuang_target_effect = {
	# 只清理已经失效的天皇目标；新目标必须由宣下奉行外交行动指定。
	if = {
		limit = {
			has_saved_global_event_target = tianhuang
			event_target:tianhuang = {
				exists = no
			}
		}
		clear_global_event_target = tianhuang
	}
	if = {
		limit = {
			has_saved_global_event_target = tianhuang
			event_target:tianhuang = {
				exists = yes
				NOT = { is_subject_of = ROOT }
			}
		}
		event_target:tianhuang = {
			clr_country_flag = mf_tianhuang_target
		}
		clear_global_event_target = tianhuang
	}
}

mf_set_xiayi_campaign_target_effect = {
	# 替换当前虾夷目标；持续战争理由由 cb_mf_xiayi_campaign 的动态条件提供。
	if = {
		limit = { has_saved_global_event_target = xiayi }
		event_target:xiayi = {
			clr_country_flag = mf_xiayi_target
		}
		clear_global_event_target = xiayi
	}
	save_global_event_target_as = xiayi
	set_country_flag = mf_xiayi_target
}

mf_refresh_xiayi_target_effect = {
	# 目标灭亡、进入幕府体系或失去虾夷标记时清理缓存，动态 CB 随条件立即失效。
	if = {
		limit = {
			has_saved_global_event_target = xiayi
			OR = {
				event_target:xiayi = { exists = no }
				event_target:xiayi = {
					OR = {
						NOT = { has_country_flag = mf_xiayi_target }
						sort_of_mf = yes
						has_country_flag = mf_tianhuang_target
					}
				}
			}
		}
		if = {
			limit = { event_target:xiayi = { exists = yes } }
			event_target:xiayi = { clr_country_flag = mf_xiayi_target }
		}
		clear_global_event_target = xiayi
	}
}
