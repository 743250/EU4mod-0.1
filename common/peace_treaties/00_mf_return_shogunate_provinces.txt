po_mf_return_shogunate_provinces = {
	# 收回没有现存幕府体系核心国的非法幕府领地。现存核心国使用原版归还核心条款。
	category = 6
	power_cost_base = 0.1
	prestige_base = 1.0
	ae_base = 0.0

	warscore_cost = {
		all_provinces = 0.0
		no_provinces = 500.0
		owner_keeps = 0.0
		conquered = 0.0
		returned_core = 0.0
		released_nation = 0.0
		cancelled_subject = 0.0
		concede_colonial_region = 0.0
	}
	warscore_cap = 80

	requires_is_allowed = yes
	applies_to_war_target = yes

	is_visible = {
		has_reform = shogunate
		FROM = {
			any_owned_province = {
				has_province_modifier = mf_unlawful_daimyo_land
				NOT = { any_core_country = { sort_of_mf = yes } }
			}
		}
	}

	is_allowed = {
		always = yes
	}

	effect = {
		FROM = {
			every_owned_province = {
				limit = {
					has_province_modifier = mf_unlawful_daimyo_land
					NOT = { any_core_country = { sort_of_mf = yes } }
				}
				remove_province_modifier = mf_unlawful_daimyo_land
				cede_province = ROOT
			}
		}
	}

	ai_weight = {
		export_to_variable = {
			variable_name = ai_value
			value = 1000
		}
	}
}
