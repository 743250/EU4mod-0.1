	# 绝罚战争的服从和平：解除绝罚、清理拒命状态，并返还幕府御统。

	power_projection = vassalized_rival
	power_cost_base = 0.25
	prestige_base = 0.5
	ae_base = 0.0

	warscore_cost = {
		all_provinces = 0.0
		no_provinces = 250.0
		owner_keeps = 0.0
		conquered = 0.0
		returned_core = 0.0
		released_nation = 0.0
		cancelled_subject = 0.0
		concede_colonial_region = 0.0
	}
	warscore_cap = 80

	requires_is_allowed = yes

	is_visible = {
		has_reform = shogunate
		FROM = {
			is_subject_of = ROOT
			OR = {
				is_subject_of_type = daimyo_vassal
				is_subject_of_type = mf_daimyo_vassal
			}
			has_country_flag = mf_under_shogunate_excommunication
		}
	}

	is_allowed = {
		always = yes
	}

	effect = {
		FROM = {
			# 清理拒绝幕府命令留下的惩戒与冷却标记。
			clr_country_flag = mf_refused_return_land
			clr_country_flag = mf_recently_refused_return_daimyo_land

			clr_country_flag = mf_refused_executor_appointment
			clr_country_flag = mf_refused_return_shogunate_province
			clr_country_flag = mf_under_shogunate_excommunication

			add_liberty_desire = -30
			add_opinion = {
				who = ROOT
				modifier = opinion_mf_forced_shogunate_obedience
			}
			every_subject_country = {
				limit = { is_subject_of_type = mf_retainer_daimyo_vassal }
				clr_country_flag = mf_refused_return_land
				clr_country_flag = mf_recently_refused_return_daimyo_land
				clr_country_flag = mf_refused_enforce_daimyo_peace
				clr_country_flag = mf_refused_partition_daimyo
				clr_country_flag = mf_refused_executor_appointment
				clr_country_flag = mf_refused_return_shogunate_province
			}
		}
		mf_change_shogun_value = {
			value = 10
		}
	}

	ai_weight = {
		export_to_variable = {
			variable_name = ai_value
			value = 1000
		}
	}
}
