po_mf_subjugate_daimyo = {
	# 大名之间的收服和平：败者和败者原有家臣都成为胜者的直属家臣。
	category = 6
	power_projection = vassalized_rival
	power_cost_base = 1.0
	prestige_base = 1.0
	ae_base = 0.5

	warscore_cost = {
		all_provinces = 1.0
		no_provinces = 0.0
		owner_keeps = 0.0
		conquered = 0.0
		returned_core = 0.0
		released_nation = 0.0
		cancelled_subject = 0.0
		concede_colonial_region = 0.0
	}
	warscore_cap = 80

	is_make_subject = yes
	requires_is_allowed = yes

	is_visible = {
		NOT = { has_reform = shogunate }
	}

	is_allowed = {
		NOT = { has_reform = shogunate }
		NOT = { is_subject_of_type = mf_retainer_daimyo_vassal }
	}

	effect = {
		if = {
			limit = { NOT = { has_reform = shogunate } }
			FROM = {
				every_subject_country = {
					limit = { is_subject_of_type = mf_retainer_daimyo_vassal }
					ROOT = {
						create_subject = {
							subject = PREV
							subject_type = mf_retainer_daimyo_vassal
						}
					}
					add_government_reform = mf_retainer_daimyo
				}
			}
			create_subject = {
				subject = FROM
				subject_type = mf_retainer_daimyo_vassal
			}
			FROM = {
				add_government_reform = mf_retainer_daimyo
			}
		}
	}

	ai_weight = {
		export_to_variable = {
			variable_name = ai_value
			value = 100
		}
	}
}
