on_startup = {
	if = {
		limit = { has_reform = shogunate }
		save_global_event_target_as = zhengyidajiangjun
		1020 = {
			set_variable = { which = shogun_value value = 50 }
			set_variable = { which = shogun_value_Profix value = 0 }
			set_variable = { which = shogun_value_display value = 50 }
			set_variable = { which = shogun_value_display_fraction value = 0 }
		}
		bl_shogun_refresh = yes
	}
}

on_monthly_pulse = {
	if = {
		limit = {
			has_reform = shogunate
			has_saved_global_event_target = zhengyidajiangjun
			tag = event_target:zhengyidajiangjun
		}
		if = {
			limit = { has_global_flag = shogun_weixin.d_global_flag }
			mf_refresh_tianhuang_target_effect = yes
		}
		if = {
			limit = { has_global_flag = shogun_weixin.i_global_flag }
			mf_refresh_xiayi_target_effect = yes
		}
		bl_mp_effect = yes
	}
}

on_siege_won_country = {
	if = {
		limit = {
			has_saved_global_event_target = zhengyidajiangjun
			has_saved_global_event_target = xiayi
			event_target:zhengyidajiangjun = { has_reform = shogunate }
			event_target:xiayi = {
				exists = yes
				has_country_flag = mf_xiayi_target
				NOT = { sort_of_mf = yes }
				NOT = { has_country_flag = mf_tianhuang_target }
			}
			ROOT = {
				OR = {
					tag = event_target:zhengyidajiangjun
					is_subject_of = event_target:zhengyidajiangjun
				}
			}
			FROM = { owned_by = event_target:xiayi }
		}
		random = {
			chance = 20
			FROM = { add_core = ROOT }
		}
	}
}

on_yearly_pulse = {
	if = {
		limit = {
			has_reform = shogunate
		}
		every_subject_country = {
			limit = {
				sort_of_mf = yes
				NOT = { has_reform = shogunate }
			}
			country_event = { id = mf_daimyo.10 }
			every_subject_country = {
				limit = {
					sort_of_mf = yes
					NOT = { has_reform = shogunate }
				}
				country_event = { id = mf_daimyo.10 }
			}
		}
	}
}

on_province_owner_change = {
	if = {
		limit = { province_id = 1020 }
		# 原版 japan.1 当日完成幕府交接，玩家新幕府次日补充自定义属国。
		owner = {
			if = {
				limit = { ai = no }
				country_event = { id = mf_daimyo.13 days = 1 }
			}
		}
	}
}
