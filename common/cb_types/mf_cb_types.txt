# 战国收服——大名内战借口，也可对家臣使用,家臣会自动让宗主接管战争
cb_mf_sengoku = {
	valid_for_subject = no
	exclusive = yes

	prerequisites_self = {
		is_revolutionary = no
		NOT = { is_subject_of_type = mf_retainer_daimyo_vassal }
		OR = {
			has_reform = daimyo
			has_reform = mf_daimyo
			has_reform = indep_daimyo
		}
	}

	prerequisites = {
		NOT = { is_subject_of_type = mf_retainer_daimyo_vassal }
		OR = {#是大名
			has_reform = daimyo
			has_reform = mf_daimyo
			has_reform = indep_daimyo
		}
		if= {
			limit = {
				is_neighbor_of = FROM
			}
		}
		else = {
			FROM = {
				any_subject_country = {
					is_subject_of_type = mf_retainer_daimyo_vassal
					is_neighbor_of = ROOT
				}
			}
		}
	}
	war_goal = mf_sengoku_subjugation
}
