########## 参勤交代 ##########
country_event = { # 年度参勤交代要求：由 on_yearly_pulse 触发。
	id = mf_daimyo.10
	title = "mf_daimyo.10.t"
	desc = "mf_daimyo.10.formula.d"
	picture = COUNTRY_COLLAPSE_eventPicture

	is_triggered_only = yes

	trigger = {
		sort_of_mf = yes
		NOT = { has_reform = shogunate }
		overlord = {
			OR = {
				AND = {
					has_reform = shogunate
					has_country_modifier = shogun_yuling_b
				}
				AND = {
					overlord = {
						has_reform = shogunate
						has_country_modifier = shogun_yuling_b
					}
				}
			}
		}
		NOT = {
			has_country_modifier = mf_sankin_kotai_travel
		}
	}

	immediate = {
		hidden_effect = {
			# 固定扣款取总发展度的五分之一。
			export_to_variable = {
				variable_name = mf_sankin_fixed_cost
				value = total_development
			}
			divide_variable = {
				which = mf_sankin_fixed_cost
				value = 5
			}
			# 距离系数取大名首都到幕府首都距离的十二分之一，最高为三。
			export_to_variable = {
				variable_name = mf_sankin_distance_multiplier
				value = capital_distance
				who = event_target:zhengyidajiangjun
			}
			divide_variable = {
				which = mf_sankin_distance_multiplier
				value = 12
			}
			if = {
				limit = {
					check_variable = {
						which = mf_sankin_distance_multiplier
						value = 3
					}
				}
				set_variable = {
					which = mf_sankin_distance_multiplier
					value = 3
				}
			}
			# 年度扣款取一年收入的三分之一，再乘以距离系数。
			export_to_variable = {
				variable_name = mf_sankin_year_cost
				value = years_of_income
			}
			divide_variable = {
				which = mf_sankin_year_cost
				value = 3
			}
			multiply_variable = {
				which = mf_sankin_year_cost
				which = mf_sankin_distance_multiplier
			}
			# 总扣款由固定扣款与年度扣款相加。
			set_variable = {
				which = mf_sankin_cost
				which = mf_sankin_fixed_cost
			}
			change_variable = {
				which = mf_sankin_cost
				which = mf_sankin_year_cost
			}
		}
	}

	option = {
		name = "mf_daimyo.10.a"
		custom_tooltip = mf_daimyo.10.formula.a.tt
		add_country_modifier = {
			name = mf_sankin_kotai_travel
			duration = 182
		}
		treasury_reduce = {
			variable = mf_sankin_cost
		}
		ai_chance = {
			factor = 100
			modifier = {
				factor = 0
				num_of_loans = 10
			}
			modifier = {
				factor = 0.1
				num_of_loans = 5
			}
			modifier = {
				factor = 0.2
				liberty_desire = 50
			}
		}
	}

	option = {
		name = "mf_daimyo.10.b"
		custom_tooltip = mf_daimyo.10.formula.b.tt
		mf_refuse_sankin_kotai_effect = yes
		ai_chance = { factor = 100 }
	}
}

province_event = { # 被恢复的大名重新纳入幕府体系。
	id = mf_daimyo.12
	title = "mf_daimyo.12.t"
	desc = "mf_daimyo.12.d"
	picture = COUNTRY_COLLAPSE_eventPicture

	is_triggered_only = yes
	hidden = yes

	immediate = {
		remove_province_modifier = mf_unlawful_daimyo_land
		owner = {
			# 归还领地可能复活已灭亡大名；若它还不在幕府下，立即重新分封。
			if = {
				limit = {
					NOT = { tag = event_target:zhengyidajiangjun }
					NOT = { is_subject_of = event_target:zhengyidajiangjun }
				}
				event_target:zhengyidajiangjun = {
					create_subject = {
						subject_type = daimyo_vassal
						subject = PREV
					}
					PREV = {
						mf_normalize_daimyo_first_layer_reform_effect = yes
					}
					mf_promote_retainer_daimyos_to_shogunate_effect = yes
				}
			}
		}
	}

	option = {
		name = "mf_daimyo.12.a"
	}
}
